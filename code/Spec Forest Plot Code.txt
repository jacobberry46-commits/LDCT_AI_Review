 library(ggplot2)
 
 data <- data.frame(
   Study = c("Wang et al. (2024)", "Warkentin et al. (2024)", "Resham et al. (2023)",
             "Huang et al. (2023)", "Venkadesh et al. (2021)", "Masquelin et al. (2021)",
             "Zhao et al. (2020)", "Li et al. (2020)"),
   FP = c(224, 296, 10, 71, 82, 46, 21, 65),
   TN = c(173, 2959, 621, 369, 736, 308, 533, 730)
 )
 
 data$spec <- data$TN / (data$TN + data$FP)
 data$spec_se <- sqrt(data$spec * (1 - data$spec) / (data$TN + data$FP))
 data$spec_lower <- data$spec - 1.96 * data$spec_se
 data$spec_upper <- data$spec + 1.96 * data$spec_se
 
 plot_data <- data.frame(
   Study = data$Study,
   Estimate = data$spec,
   Lower = data$spec_lower,
   Upper = data$spec_upper,
   Metric = "Specificity"
 )
 
 pooled <- data.frame(
   Study = "Pooled",
   Estimate = 0.898,
   Lower = 0.787,
   Upper = 0.955,
   Metric = "Specificity"
 )
 
 final_plot_data <- rbind(plot_data, pooled)
 study_levels <- c(unique(data$Study), "Pooled")
 final_plot_data$Study <- factor(final_plot_data$Study, levels = rev(study_levels))
 final_plot_data$Label <- sprintf("%.3f (%.3f–%.3f)", final_plot_data$Estimate,
                                  final_plot_data$Lower, final_plot_data$Upper)
 
 study_positions <- data.frame(
   Study = levels(final_plot_data$Study),
   y = seq_along(levels(final_plot_data$Study))
 )
 pooled_row <- merge(final_plot_data[final_plot_data$Study == "Pooled", ], study_positions, by = "Study")
 
 make_diamond <- function(center, lower, upper, y_level) {
   data.frame(
     x = c(lower, center, upper, center),
     y = c(y_level, y_level + 0.2, y_level, y_level - 0.2)
   )
 }
 
 diamond_data <- make_diamond(pooled_row$Estimate, pooled_row$Lower, pooled_row$Upper, pooled_row$y)
 diamond_data$Metric <- "Specificity"
 

 ggplot(final_plot_data, aes(x = Estimate, y = Study)) +
   geom_vline(xintercept = 0.898, color = "red", linetype = "dotted", linewidth = 0.8) +
   geom_point(data = subset(final_plot_data, Study != "Pooled"),
              color = "black", size = 3) +
   geom_errorbarh(data = subset(final_plot_data, Study != "Pooled"),
                  aes(xmin = Lower, xmax = Upper), color = "black", height = 0.2) +
   geom_polygon(data = diamond_data, aes(x = x, y = y), fill = "black", color = "black", alpha = 0.6) +
   geom_text(aes(x = 1.03, label = Label), hjust = 0, size = 3.5) +
   scale_y_discrete(limits = rev(study_levels)) +
   coord_cartesian(xlim = c(0.4, 1.1)) +
   theme_minimal() +
   theme(
     panel.grid = element_blank(),
     legend.position = "none",
     plot.title = element_blank(),
     strip.text = element_blank(),
     axis.title.y = element_blank(),
     axis.text.x = element_text(size = 8),
     axis.text.y = element_text(size = 9),
     panel.spacing = unit(0.5, "lines"),
     plot.margin = margin(5, 30, 5, 5)
   ) +
   labs(x = "Specificity")
